    class Config:
    """Reads a run's YAML config and hands its values to library functions.

    This is the only place that knows the YAML layout. Library functions take plain
    arguments (``run_dir``, ``ppo_params``, ...), so a different config layout only
    needs a subclass that overrides these properties.
    """

    def __init__(self, config: Path):
        self.cfg = read_config(config)

    def load(self, config):
        self.cfg = read_config(config)
        return self.cfg

    def save(self, path: Path = None, make_dir=True, rewrite_test=True, warn_github_changes=True):
        """Write the config to ``path`` (default ``run_dir / 'config.yaml'``).

        Runs not named ``test`` record the current git commit as ``git_hash``.
        """
        if path is None:
            path = self.run_dir / 'config.yaml'
        if make_dir:
            os.makedirs(self.run_dir, exist_ok=self.is_test)

        if not self.is_test and rewrite_test:
            self.cfg.git_hash = get_commit_hash(warn=warn_github_changes)

        save_config(self.cfg, path)

    @property
    def algorithm(self) -> str:
        """Learning algorithm, e.g. ``'ppo'`` (the default when unset)."""
        return self.cfg.get('algorithm', 'ppo')

    @property
    def env_name(self) -> str:
        """Environment class name, e.g. ``'Cheetah'``."""
        return self.cfg.env

    @property
    def backend(self) -> str:
        """Backend for evaluation environments, ``'np'`` or ``'jnp'``."""
        return self.cfg.backend

    @property
    def env_params(self) -> config_dict.ConfigDict:
        """The ``env_config`` section, given to env classes as ``env_params``."""
        return create_config_dict(self.cfg.env_config.to_dict())

    @property
    def ppo_params(self) -> dict:
        """Brax PPO settings, passed to the trainer as keyword arguments."""
        return self.cfg.learning_params.ppo_params.to_dict()

    @property
    def network_params(self) -> dict:
        """Network settings for the configured algorithm."""
        return self.cfg.learning_params.network_params.to_dict()

    @property
    def normalize_observations(self) -> bool:
        """Whether training keeps a running observation normalizer."""
        return self.cfg.learning_params.ppo_params.normalize_observations

    @property
    def name(self) -> str:
        """Run name."""
        return self.cfg.name

    @property
    def total_timesteps(self) -> str:
        return self.cfg.learning_params.ppo_params.num_timesteps

    @name.setter
    def name(self, value: str):
        self.cfg.name = str(value)

    @property
    def is_test(self) -> bool:
        """Runs named ``test`` may be overwritten and do not record a git hash."""
        return self.name == 'test'

    @property
    def save_dir(self) -> Path:
        """Parent directory of all runs for this config."""
        return Path(self.cfg.save_dir)

    @save_dir.setter
    def save_dir(self, value):
        self.cfg.save_dir = str(value)

    @property
    def run_dir(self) -> Path:
        """Run directory: ``save_dir / name``."""
        return self.save_dir / self.name

    def get_commit_hash(warn=True):
    import subprocess

    try:
        commit_hash = subprocess.check_output(
            ['git', 'rev-parse', 'HEAD']
        ).decode('utf-8').strip()

        # Check for unadded / uncommitted changes
        status_output = subprocess.check_output(
            ['git', 'status', '--porcelain']
        ).decode('utf-8').strip()

        if status_output and warn:
            input("⚠️ Warning: There are unadded or uncommitted changes in the repository. Press ENTER to continue...")

        return commit_hash

    except subprocess.CalledProcessError as e:
        print(f"Error getting commit hash: {e}")
        return None

    def flatten_config(config, parent_key='', sep='/'):
    """Flatten a nested (dict-like) config so every leaf is keyed by its path, joined by `sep`.
    """
    items = {}
    for key, value in config.items():
        key = str(key)
        if sep in key:
            raise ValueError(
                f"Config key '{key}'{f' (under {parent_key})' if parent_key else ''} contains "
                f"'{sep}', which is reserved as the delimiter between nesting levels. Rename "
                f"the key (e.g. '{key.replace(sep, '_')}') or flatten with a different `sep`."
            )
        new_key = f'{parent_key}{sep}{key}' if parent_key else key
        if hasattr(value, 'items'):
            items.update(flatten_config(value, new_key, sep=sep))
        else:
            items[new_key] = value
    return items

    def unflatten_config(flat_config, sep='/'):
    """Inverse of `flatten_config`: rebuild a nested config from its `a/b/c` keys.
    """
    config = {}
    for key, value in flat_config.items():
        if key.startswith('_'):
            continue
        *parents, leaf = key.split(sep)
        node = config
        for depth, parent in enumerate(parents):
            node = node.setdefault(parent, {})
            if not isinstance(node, dict):
                raise ValueError(
                    f"Cannot unflatten '{key}': '{sep.join(parents[:depth + 1])}' is a "
                    f"value ({node!r}), not a section."
                )
        node[leaf] = value
    return config

    def save_model(current_step, make_policy, params, network_config, output_dir: Path, run: wandb.Run = None):
    """Save a Brax checkpoint and optionally log it as a W&B artifact."""
    checkpoint.save(
        path=output_dir.resolve(),
        step=current_step,
        params=params,

    def get_earliest_artifact(
    run: wandb.apis.public.Run, prefix: str, iterations: float
) -> wandb.Artifact:
    """Return the first artifact of `run` matching `prefix` written at or past
    `iterations` training steps, or raise ValueError.

    A checkpoint artifact records the step it was written at in `metadata['iteration']`,

    """
    matches = [
        artifact for artifact in run.logged_artifacts()
        if prefix in artifact.name and 'iteration' in artifact.metadata
    ]
    past = [a for a in matches if a.metadata['iteration'] >= iterations]

            f"{run.id}. Logged iterations: "
            f"{sorted(a.metadata['iteration'] for a in matches) or 'none'}"
        )
    return min(past, key=lambda a: a.metadata['iteration'])

        self.run                = run
        self.times              = []
        self.x_data             = []
        self.y_data             = []
        self.y_dataerr          = []

    def plot_progress(self, num_steps, metrics) -> None:
        """Brax PPO progress callback: accumulate eval reward, log native W&B lines.

        Logs every scalar brax metric via ``run.log`` so each renders as a native,
        interactive W&B line chart, and writes the raw curve to ``progress.csv``.
        """
        print('=== TRAINING EPOCH ===')
        print('time', datetime.now(ZoneInfo("America/New_York")).strftime("%Y-%m-%d %H:%M:%S %Z"))
        print('num_steps', num_steps)
        if self.total_timesteps is not None:
            print('total steps', self.total_timesteps)

        self.times.append(datetime.now())
        self.x_data.append(int(num_steps))
        self.y_data.append(float(metrics["eval/episode_reward"]))
        self.y_dataerr.append(float(metrics["eval/episode_reward_std"]))
        pd.DataFrame(
            {
                'times': self.times,
                'x': self.x_data,
                'y': self.y_data,
                'yerr': self.y_dataerr
            }
        ).to_csv(
            self.save_dir / 'progress.csv',
            index=False
        )

        if np.isnan(self.y_data[-1]) or np.isnan(self.y_dataerr[-1]):
            raise Exception(f'NaN found... \n\n{self.y_data}\n\n{self.y_dataerr}')

        if self.run:
            # Log every scalar brax metric so each gets a native line chart.
            scalars = {}
            for k, v in metrics.items():
                try:
                    scalars[k] = float(v)
                except (TypeError, ValueError):
                    continue
            self.run.log(scalars, step=num_steps)

                f"Delete this run and its {artifact_count} artifact(s)?"
            ):
                print("  - Skipped.")
                print("-" * 40)
                skipped_count += 1
                continue

            print("  - Deleting...")
            print("-" * 40)

            # Deletes the run and all of its associated artifacts
            run.delete(delete_artifacts=True)
            deleted_count += 1

    print(
        f"\nExecution finished. Successfully deleted {deleted_count} runs "
        f"({skipped_count} skipped)."
    )
